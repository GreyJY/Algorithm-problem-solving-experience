import java.util.Scanner;
import java.math.BigInteger;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();

        BigInteger[] f = new BigInteger[n + 1];
        f[1] = BigInteger.ONE;

        for (int i = 2; i <= n; i++) {
            f[i] = f[i - 1];
            if (i % 2 == 0) {
                f[i] = f[i].add(f[i / 2]);先把最开始的数字为2的合法数列的个数，遵循奇数数量跟偶数数量一样多，偶数数量等于奇数数量加这个偶数本身的二分之一的数字的数量
            }
        }

        System.out.println(f[n]);
    }
}
找到共同之处，f[n]表示的是输入i数字情况下合法数列有几个.用到递推思想：先想到最基础最小的数字情况，再慢慢递推到最大的情况，或者哪边好推推哪边
不添加：       [6]

下一项是 1：  [6, 1]

下一项是 2：  [6, 2]
              [6, 2, 1]

下一项是 3：  [6, 3]
              [6, 3, 1]
