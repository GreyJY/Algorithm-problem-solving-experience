import java.util.Scanner;
import java.math.BigInteger;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();

        BigInteger[] f = new BigInteger[n + 1];
        f[1] = BigInteger.ONE;

        for (int i = 2; i <= n; i++) {
            f[i] = f[i - 1];
            if (i % 2 == 0) {
                f[i] = f[i].add(f[i / 2]);
            }
        }

        System.out.println(f[n]);
    }
}
