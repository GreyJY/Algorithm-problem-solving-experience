排序器可以这样写
 Arrays.sort(students, (a, b) -> {
            // 总分不同：总分高的在前
            if (a[2] != b[2]) {
                return b[2] - a[2];//减号作为排序的结果，这是从大到小
            }

            // 总分相同、语文不同：语文高的在前
            if (a[1] != b[1]) {
                return b[1] - a[1];//加if条件来写
            }

            // 总分和语文都相同：学号小的在前
            return a[0] - b[0];
        });
